// docker related code to buid, execute docker locally
cd D:\MySampleSpringBootProject\SampleMicroservices_SpringBoot\service-registry\ // got project service directory
mvn clean package -DskipTests // to create jar file for springboot application

docker build -t service-registry . // to build/ generate image 
docker build -t micro-service-registry . // to build/ generate image  (name change)
docker build -t micro-service-registry:latest . // to build/ generate image  (with tag latest)
docker build -t micro-service-registry:05d3910 . // to build/ generate image  (with custom tag)
docker build -t micro-service-registry:05d3910 -t micro-service-registry:latest . // to build/ generate image  (with multiple tags)

docker run -p 8080:8080 micro-service-registry // to select image and run container for that image
docker run -e PORT=8080 -p 8080:8080 micro-service-registry // to select image and run container for that image (pass env var)
docker run -e PORT=8080 -p 8080:8080 micro-service-registry:05d3910 // to select image and run container for that image (select image with specfic tag)
docker run -e PORT=8080 -p 8080:8080 micro-service-registry:latest // to select image and run container for that image (select image with latest tag)

docker stop containerId // to stop container


docker network create microservices-network
